// Example structure to place inside your shop.html script tags function addToCart(id, name, price) { let cart = JSON.parse(localStorage.getItem('cart')) || []; // Check if item already exists in cart state let existingItem = cart.find(item => item.id === id); if (existingItem) { existingItem.quantity += 1; } else { cart.push({ id: id, name: name, price: price, quantity: 1 }); } // Push modern array directly up to localStorage string database localStorage.setItem('cart', JSON.stringify(cart)); alert(`${name} added to bag!`); }
